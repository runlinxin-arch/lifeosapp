**锁定的行为契约：**
1. `title` 非空（去首尾空白）；`type` / `status` 受枚举约束
2. `totalUnits` / `doneUnits` 为非负整数；`doneUnits` 不得超过 `totalUnits`（`totalUnits` 为 0 时视为"未设目标"，`doneUnits` 必须为 0）
3. `startDate` / `endDate` 同时给出时才校验区间（`endDate >= startDate`）；两者都可空
4. 状态为 `done` 时，若 `totalUnits > 0` 则进度补满
5. `progressPercent`：`totalUnits <= 0` 返回 0，否则 `round(done/total*100)` 并夹到 0–100
6. `createSession` 的 `minutes` 必须是 ≥1 的整数；`date` 必须是合法日期
7. `deleteLearnItem` 在同一事务里删除学习项并调用 `removeBySource(db, 'learning', id)`

```ts
import type { DatabaseSync } from 'node:sqlite'
import type {
  LearnItem, LearnItemInput, LearnItemPatch, LearnNote, LearnSession, LearnStatus, LearnType
} from '@shared/types'
import { assertDateStr, assertNonEmpty } from './validate'
import { removeBySource } from './longTasks'

const TYPES: readonly LearnType[] = ['book', 'course', 'skill']
const STATUSES: readonly LearnStatus[] = ['not_started', 'learning', 'done']
/** 列表展示顺序：学习中 → 未开始 → 已完成 */
const STATUS_ORDER: Record<LearnStatus, number> = { learning: 0, not_started: 1, done: 2 }

// eslint-disable-next-line @typescript-eslint/no-explicit-any
const rowToItem = (r: any): LearnItem => r as LearnItem

export function progressPercent(item: Pick<LearnItem, 'total_units' | 'done_units'>): number {
  if (!item.total_units || item.total_units <= 0) return 0
  const pct = Math.round((item.done_units / item.total_units) * 100)
  return Math.max(0, Math.min(100, pct))
}

function assertUnits(totalUnits: number | undefined, doneUnits: number | undefined): void {
  if (totalUnits !== undefined) {
    if (!Number.isInteger(totalUnits) || totalUnits < 0) throw new Error('总量必须是不小于 0 的整数')
  }
  if (doneUnits !== undefined) {
    if (!Number.isInteger(doneUnits) || doneUnits < 0) throw new Error('进度必须是不小于 0 的整数')
    const total = totalUnits ?? 0
    if (doneUnits > total) throw new Error(`进度不能超过总量（总量 ${total}）`)
    if (total === 0 && doneUnits > 0) throw new Error('未设置总量时进度必须为 0')
  }
}

function assertType(t: string): void {
  if (!TYPES.includes(t as LearnType)) throw new Error(`学习项类型只能是 book / course / skill，收到：${t}`)
}

function assertStatus(s: string): void {
  if (!STATUSES.includes(s as LearnStatus)) {
    throw new Error(`学习项状态只能是 not_started / learning / done，收到：${s}`)
  }
}

/** 两端都给出时才校验区间；单端可空 */
function assertDates(start: string | null | undefined, end: string | null | undefined): void {
  if (start) assertDateStr(start, '开始日期')
  if (end) assertDateStr(end, '结束日期')
  if (start && end && end < start) throw new Error('结束日期不能早于开始日期')
}

export function listLearnItems(db: DatabaseSync): LearnItem[] {

export function updateLearnItem(db: DatabaseSync, id: number, patch: LearnItemPatch): LearnItem {
  const cur = getLearnItem(db, id)
  if (!cur) throw new Error(`learn_item ${id} not found`)

  const title = patch.title !== undefined ? assertNonEmpty(patch.title, '学习项标题') : cur.title
  const type = patch.type ?? cur.type
  const status = patch.status ?? cur.status
  assertType(type)
  assertStatus(status)

  const start = patch.startDate !== undefined ? patch.startDate : cur.start_date
  const end = patch.endDate !== undefined ? patch.endDate : cur.end_date
  assertDates(start, end)

  const total = patch.totalUnits ?? cur.total_units
  let done = patch.doneUnits ?? cur.done_units
  assertUnits(total, done)
  // 状态改为已完成且设了总量时，进度补满
  if (status === 'done' && total > 0) done = total
  if (done > total) throw new Error(`进度不能超过总量（总量 ${total}）`)

  db.prepare(`
    UPDATE learn_items SET title=?, type=?, status=?, total_units=?, done_units=?,
      start_date=?, end_date=?, updated_at=datetime('now','localtime')
    WHERE id=?
  `).run(title, type, status, total, done, start, end, id)
  return getLearnItem(db, id)!
}
```

```ts
import type { LearnSession } from './types'
import { addDays } from './dates'
import { weekDays } from './layout'

/** 含端点的时长合计 */
export function sumMinutes(sessions: LearnSession[], from: string, to: string): number {
  let total = 0
  for (const s of sessions) if (s.date >= from && s.date <= to) total += s.minutes
  return total
}

/** 按给定日期序列汇总（缺失记 0，顺序与入参一致，可直接喂柱状图） */
export function dailyMinutes(sessions: LearnSession[], days: string[]): { date: string; minutes: number }[] {
  const byDate = new Map<string, number>()
  for (const s of sessions) byDate.set(s.date, (byDate.get(s.date) ?? 0) + s.minutes)
  return days.map((date) => ({ date, minutes: byDate.get(date) ?? 0 }))
}

/**
 * 连续学习天数：
 * 今天有记录 → 从今天往前连数；今天没有但昨天有 → 从昨天往前连数（连续尚未中断）；否则 0
 */
export function streakDays(sessions: LearnSession[], today: string): number {
  const days = new Set(sessions.map((s) => s.date))
  if (days.size === 0) return 0
  let cursor = days.has(today) ? today : addDays(today, -1)
  if (!days.has(cursor)) return 0
  let count = 0
  while (days.has(cursor)) {
    count++
    cursor = addDays(cursor, -1)
  }
  return count
}

/** 本周（周一~周日）合计时长 */
export function weekMinutes(sessions: LearnSession[], today: string): number {
  const days = weekDays(today)
  return sumMinutes(sessions, days[0], days[6])
}
```

```ts
import { create } from 'zustand'
import type { LearnItem, LearnItemInput, LearnItemPatch, LearnNote, LearnSession } from '@shared/types'
import { api } from '../lib/ipc'
import { todayStr, addDays } from '../lib/dates'
import { formatIpcError } from '../lib/errors'

interface LearnState {
  items: LearnItem[]
  notes: LearnNote[]
  /** 最近 400 天的学习记录，供统计与连续天数使用 */
  sessions: LearnSession[]
  selectedId: number | null
  loading: boolean
  error: string | null
  clearError: () => void
  load: () => Promise<boolean>
  select: (id: number | null) => void
  createItem: (input: LearnItemInput) => Promise<boolean>
  updateItem: (id: number, patch: LearnItemPatch) => Promise<boolean>
  removeItem: (id: number) => Promise<boolean>
  setProgress: (id: number, doneUnits: number) => Promise<boolean>
  createNote: (itemId: number, content: string) => Promise<boolean>
  updateNote: (id: number, content: string) => Promise<boolean>
  removeNote: (id: number) => Promise<boolean>
  createSession: (itemId: number, date: string, minutes: number) => Promise<boolean>
  removeSession: (id: number) => Promise<boolean>
}

export const useLearnStore = create<LearnState>((set, get) => {
  const run = async (action: () => Promise<void>): Promise<boolean> => {
    try {
      await action()
      set({ error: null })
      return true
    } catch (e) {
      set({ error: formatIpcError(e) })
      return false
    }
  }

  const fetchItems = async (): Promise<void> => set({ items: await api.learn.listItems() })
  const fetchSessions = async (): Promise<void> => {
    const from = addDays(todayStr(), -400)
    set({ sessions: await api.learn.listSessions(from, todayStr()) })
  }
  const fetchNotes = async (itemId: number | null): Promise<void> => {
    set({ notes: itemId === null ? [] : await api.learn.listNotes(itemId) })
  }

  return {
    items: [],
    notes: [],
    sessions: [],
    selectedId: null,
    loading: false,
    error: null,
    clearError: () => set({ error: null }),

    load: () => run(async () => {
      set({ loading: true })
      await Promise.all([fetchItems(), fetchSessions()])
      const { selectedId, items } = get()
      // 选中的项被删掉时自动退回未选中
      const stillExists = selectedId !== null && items.some((i) => i.id === selectedId)
      await fetchNotes(stillExists ? selectedId : null)
      if (!stillExists) set({ selectedId: null })
      set({ loading: false })
    }),

    select: (id) => {
      set({ selectedId: id })
      void run(() => fetchNotes(id))
    },

    createItem: (input) => run(async () => { await api.learn.createItem(input); await fetchItems() }),
    updateItem: (id, patch) => run(async () => { await api.learn.updateItem(id, patch); await fetchItems() }),
    removeItem: (id) => run(async () => {
      await api.learn.removeItem(id)
      if (get().selectedId === id) set({ selectedId: null, notes: [] })
      await fetchItems()
    }),
    setProgress: (id, doneUnits) => run(async () => { await api.learn.setProgress(id, doneUnits); await fetchItems() }),
    createNote: (itemId, content) => run(async () => { await api.learn.createNote(itemId, content); await fetchNotes(itemId) }),
    updateNote: (id, content) => run(async () => { await api.learn.updateNote(id, content); await fetchNotes(get().selectedId) }),
    removeNote: (id) => run(async () => { await api.learn.removeNote(id); await fetchNotes(get().selectedId) }),
    createSession: (itemId, date, minutes) => run(async () => {
      await api.learn.createSession(itemId, date, minutes)
      await fetchSessions()
    }),
    removeSession: (id) => run(async () => { await api.learn.removeSession(id); await fetchSessions() })
  }
})
```

```tsx
import { useEffect, useRef, useState } from 'react'
import { useLearnStore } from '../../stores/learnStore'
import { useToday } from '../../hooks/useToday'

/** 秒 → 展示用 mm:ss / h:mm:ss */
function formatElapsed(seconds: number): string {
  const h = Math.floor(seconds / 3600)
  const m = Math.floor((seconds % 3600) / 60)
  const s = seconds % 60
  const pad = (n: number): string => String(n).padStart(2, '0')
  return h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`
}

/** 已计秒数 → 入库分钟数（至少 1 分钟） */
export function elapsedToMinutes(seconds: number): number {
  return Math.max(1, Math.round(seconds / 60))
}

export function LearnTimer({ itemId }: { itemId: number }): JSX.Element {
  const createSession = useLearnStore((s) => s.createSession)
  const today = useToday()
  const [startedAt, setStartedAt] = useState<number | null>(null)
  const [elapsed, setElapsed] = useState(0)
  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null)

  useEffect(() => {
    if (startedAt === null) {
      if (tickRef.current) clearInterval(tickRef.current)
      tickRef.current = null
      return
    }
    tickRef.current = setInterval(() => {
      setElapsed(Math.floor((Date.now() - startedAt) / 1000))
    }, 1000)
    return () => { if (tickRef.current) clearInterval(tickRef.current) }
  }, [startedAt])

  // 切换学习项时丢弃未结束的计时
  useEffect(() => { setStartedAt(null); setElapsed(0) }, [itemId])

  const start = (): void => { setElapsed(0); setStartedAt(Date.now()) }

  const stop = async (): Promise<void> => {
    if (startedAt === null) return
    const seconds = Math.floor((Date.now() - startedAt) / 1000)
    setStartedAt(null)
    setElapsed(0)
    await createSession(itemId, today, elapsedToMinutes(seconds))
  }

  return (
    <div className="bg-[var(--card)] border border-[var(--border)] rounded-lg p-3 mb-3">
      <div className="text-xs text-[var(--muted)] mb-2">学习计时</div>
      <div className="flex items-center gap-3">
        <span className="text-2xl font-mono tabular-nums" style={{ color: startedAt ? '#4ade80' : undefined }}>
          {formatElapsed(elapsed)}
        </span>
        {startedAt === null
          ? <button onClick={start} className="text-xs px-3 py-1.5 rounded bg-[var(--accent)] text-white">开始</button>
          : <button onClick={() => void stop()} className="text-xs px-3 py-1.5 rounded bg-red-500/20 text-red-300">停止并记录</button>}
        <span className="text-[10px] text-[var(--muted)] ml-auto">停止时记录一条学习时长（至少 1 分钟）</span>
      </div>
    </div>
  )
}
```

```tsx
import type { LearnItem } from '@shared/types'
import { useLongTasksStore } from '../../stores/longTasksStore'
import { useLearnStore } from '../../stores/learnStore'

export function LearnMarkButton({ item }: { item: LearnItem }): JSX.Element {
  const { tasks, markFromSource, unmarkSource, error, clearError } = useLongTasksStore()
  const error2 = useLearnStore((s) => s.error)

    await markFromSource({

      startDate: item.start_date!,
      endDate: item.end_date!,
      sourceModule: 'learning',
      sourceId: item.id
    })
  }

    <div className="bg-[var(--card)] border border-[var(--border)] rounded-lg p-3 mb-3">
      <div className="text-xs text-[var(--muted)] mb-2">长期任务</div>
      <div className="flex items-center gap-2">

          <>
            <span className="text-xs px-2 py-0.5 rounded" style={{ background: '#4ade80', color: '#0b1f12' }}>
              已上轴
            </span>
            <button
              onClick={() => { void unmarkSource('learning', item.id) }}
              className="text-xs px-2 py-1 rounded bg-[var(--border)]"
            >取消标记</button>
          </>
        ) : (
          <button
            onClick={() => void mark()}
            disabled={!hasRange}
            className={`text-xs px-3 py-1.5 rounded ${hasRange ? 'bg-[var(--accent)] text-white' : 'bg-[var(--border)] text-[var(--muted)] cursor-not-allowed'}`}
          >标记为长期任务</button>
        )}

          <span className="flex-1">{error ?? error2}</span>
          <button onClick={() => { clearError() }} className="text-[var(--muted)]">✕</button>
        </div>
      )}
    </div>
  )
}
```
