```ts
export function toDateStr(d: Date): string {
  const y = d.getFullYear()
  const m = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  return `${y}-${m}-${day}`
}

export function todayStr(): string {
  return toDateStr(new Date())
}

export function addDays(dateStr: string, n: number): string {
  const d = new Date(dateStr + 'T00:00:00')
  d.setDate(d.getDate() + n)
  return toDateStr(d)
}

export function diffDays(a: string, b: string): number {
  const ms = new Date(a + 'T00:00:00').getTime() - new Date(b + 'T00:00:00').getTime()
  return Math.round(ms / 86400000)
}

export function formatCn(dateStr: string): string {
  const [, m, d] = dateStr.split('-')
  return `${Number(m)}/${Number(d)}`
}
```

```ts
import Database from 'better-sqlite3'
import path from 'node:path'
import fs from 'node:fs'

export function openDb(file: string): Database.Database {
  const db = new Database(file)
  db.pragma('journal_mode = WAL')
  db.pragma('foreign_keys = ON')
  migrate(db)
  return db
}

export function dataDir(): string {
  // 主进程内通过 app.getPath('userData') 调用方注入；测试传临时目录
  return process.env.LIFEOS_DATA_DIR ?? ''
}

export function ensureDataDir(dir: string): string {
  fs.mkdirSync(path.join(dir, 'backups'), { recursive: true })
  return dir
}

function migrate(db: Database.Database): void {
  db.exec(`
    CREATE TABLE IF NOT EXISTS tasks (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      done INTEGER NOT NULL DEFAULT 0,
      priority TEXT NOT NULL DEFAULT 'mid' CHECK (priority IN ('low','mid','high')),
      done_at TEXT,
      date TEXT NOT NULL,
      source_module TEXT,
      source_id INTEGER,
      created_at TEXT NOT NULL DEFAULT (datetime('now','localtime')),
      updated_at TEXT NOT NULL DEFAULT (datetime('now','localtime'))
    );
    CREATE INDEX IF NOT EXISTS idx_tasks_date ON tasks(date);
    CREATE TABLE IF NOT EXISTS long_tasks (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      start_date TEXT NOT NULL,
      end_date TEXT NOT NULL,
      color TEXT NOT NULL DEFAULT '#8b8d98',
      source_module TEXT,
      source_id INTEGER,
      created_at TEXT NOT NULL DEFAULT (datetime('now','localtime')),
      updated_at TEXT NOT NULL DEFAULT (datetime('now','localtime'))
    );
    CREATE INDEX IF NOT EXISTS idx_long_tasks_range ON long_tasks(start_date, end_date);
    CREATE TABLE IF NOT EXISTS settings (
      key TEXT PRIMARY KEY,
      value TEXT NOT NULL
    );
  `)
}
```

`src/main/db/tasks.ts`：

```ts
import type { Database } from 'better-sqlite3'
import type { Priority, Task } from '@shared/types'

export interface TaskInput {
  title: string
  priority?: Priority
  date: string
}

export type TaskPatch = Partial<Pick<Task, 'title' | 'done' | 'priority'>>

const rowToTask = (r: Record<string, unknown>): Task =>
  ({ ...(r as object), done: !!r.done }) as Task

export function listTasksByDate(db: Database, date: string): Task[] {
  return db.prepare(`
    SELECT * FROM tasks WHERE date = ?
    ORDER BY done ASC,
      CASE priority WHEN 'high' THEN 0 WHEN 'mid' THEN 1 ELSE 2 END,
      id ASC
  `).all(date).map(rowToTask)
}

export function createTask(db: Database, input: TaskInput): Task {
  const info = db.prepare(
    'INSERT INTO tasks (title, priority, date) VALUES (?, ?, ?)'
  ).run(input.title, input.priority ?? 'mid', input.date)
  return getTask(db, Number(info.lastInsertRowid))!
}

export function getTask(db: Database, id: number): Task | null {
  const row = db.prepare('SELECT * FROM tasks WHERE id = ?').get(id)
  return row ? rowToTask(row) : null
}

export function updateTask(db: Database, id: number, patch: TaskPatch): Task {
  const cur = getTask(db, id)
  if (!cur) throw new Error(`task ${id} not found`)
  const title = patch.title ?? cur.title
  const priority = patch.priority ?? cur.priority
  const done = patch.done ?? cur.done
  const doneAt = done ? (cur.done_at ?? new Date().toISOString()) : null
  db.prepare(`
    UPDATE tasks SET title=?, priority=?, done=?, done_at=?, updated_at=datetime('now','localtime')
    WHERE id=?
  `).run(title, priority, done ? 1 : 0, doneAt, id)
  return getTask(db, id)!
}

export function deleteTask(db: Database, id: number): void {
  db.prepare('DELETE FROM tasks WHERE id = ?').run(id)
}
```

```ts
import type { Database } from 'better-sqlite3'
import type { LongTask } from '@shared/types'

export interface LongTaskInput {
  title: string
  startDate: string
  endDate: string
  color?: string
}

const rowToLongTask = (r: Record<string, unknown>): LongTask => r as unknown as LongTask

export function listRange(db: Database, from: string, to: string): LongTask[] {
  return db.prepare(`
    SELECT * FROM long_tasks
    WHERE start_date <= ? AND end_date >= ?
    ORDER BY start_date ASC, id ASC
  `).all(to, from).map(rowToLongTask)
}

export function createLongTask(db: Database, input: LongTaskInput): LongTask {
  const info = db.prepare(`
    INSERT INTO long_tasks (title, start_date, end_date, color)
    VALUES (?, ?, ?, ?)
  `).run(input.title, input.startDate, input.endDate, input.color ?? '#8b8d98')
  const row = db.prepare('SELECT * FROM long_tasks WHERE id = ?').get(Number(info.lastInsertRowid))
  return rowToLongTask(row as Record<string, unknown>)
}

export function deleteLongTask(db: Database, id: number): void {
  db.prepare('DELETE FROM long_tasks WHERE id = ?').run(id)
}
```

```ts
import type { Task, Priority } from '@shared/types'
import { api } from '../lib/ipc'
import { todayStr } from '../lib/dates'

interface TasksState {
  tasks: Task[]
  load: (date: string) => Promise<void>
  add: (title: string, priority: Priority) => Promise<void>
  toggle: (id: number) => Promise<void>
  remove: (id: number) => Promise<void>
}

export const useTasksStore = create<TasksState>((set, get) => ({
  tasks: [],
  load: async (date) => {
    const tasks = await api.tasks.list(date)
    set({ tasks })
  },
  add: async (title, priority) => {
    await api.tasks.create({ title, priority, date: todayStr() })
    await get().load(todayStr())
  },
  toggle: async (id) => {
    const t = get().tasks.find((x) => x.id === id)
    if (!t) return
    await api.tasks.update(id, { done: !t.done })
    await get().load(todayStr())
  },
  remove: async (id) => {
    await api.tasks.remove(id)
    await get().load(todayStr())
  }
}))
```

```ts
import type { LongTask } from '@shared/types'
import { diffDays } from './dates'

export interface TimelineBlock {
  id: number
  title: string
  color: string
  leftPct: number
  widthPct: number
  lane: number
}

export function calcBlocks(tasks: LongTask[], from: string, to: string): TimelineBlock[] {
  const total = diffDays(to, from) + 1
  const sorted = [...tasks].sort((a, b) => a.start_date.localeCompare(b.start_date))
  const laneEnds: string[] = []
  return sorted.map((t) => {
    const s = t.start_date < from ? from : t.start_date
    const e = t.end_date > to ? to : t.end_date
    let lane = laneEnds.findIndex((end) => end < s)
    if (lane === -1) { lane = laneEnds.length; laneEnds.push('') }
    laneEnds[lane] = t.end_date
    return {
      id: t.id,
      title: t.title,
      color: t.color,
      leftPct: (diffDays(s, from) / total) * 100,
      widthPct: ((diffDays(e, s) + 1) / total) * 100,
      lane
    }
  }).filter((b) => b.widthPct > 0)
}
```

```tsx
  const { tasks, range, reload, create, remove } = useLongTasksStore()
  const [expanded, setExpanded] = useState(false)
  const [selected, setSelected] = useState<LongTask | null>(null)
  const [creating, setCreating] = useState(false)
  const [form, setForm] = useState({ title: '', start: todayStr(), end: addDays(todayStr(), 13) })

  useEffect(() => { void reload() }, [reload])

  const blocks = useMemo(() => calcBlocks(tasks, range.from, range.to), [tasks, range])
  const laneCount = Math.max(1, ...blocks.map((b) => b.lane + 1))
  const today = todayStr()
  const todayPct = (diffDays(today, range.from) / (diffDays(range.to, range.from) + 1)) * 100
  const ticks = useMemo(() => {
    const list: { leftPct: number; label: string }[] = []
    for (let d = range.from; d <= range.to; d = addDays(d, 7)) {
      list.push({ leftPct: (diffDays(d, range.from) / (diffDays(range.to, range.from) + 1)) * 100, label: formatCn(d) })
    }
    return list
  }, [range])

  const submitCreate = async (): Promise<void> => {
    if (!form.title.trim()) return
    await create(form.title.trim(), form.start, form.end)
    setCreating(false)
    setForm({ title: '', start: todayStr(), end: addDays(todayStr(), 13) })
  }

  const barHeight = expanded ? 14 : 8
  const rowHeight = barHeight + 4

  return (
    <div className="flex items-center gap-3 w-full px-2">
      <button onClick={() => setExpanded(!expanded)} className="text-[var(--muted)] hover:text-[var(--text)] text-xs whitespace-nowrap">
        {expanded ? '收起 ⤡' : '展开 ⤢'}
      </button>
      <div className="flex-1 relative select-none">
        {ticks.map((t, i) => (
          <span key={i} className="absolute top-0 text-[9px] text-[var(--muted)] -translate-x-1/2" style={{ left: `${t.leftPct}%` }}>{t.label}</span>
        ))}
        <div className="relative w-full bg-[var(--border)] rounded" style={{ height: laneCount * rowHeight + 6, marginTop: 14 }}>
          {blocks.map((b) => (
            <div
              key={b.id}
              onClick={() => setSelected(tasks.find((t) => t.id === b.id) ?? null)}
              title={expanded ? undefined : b.title}
              className="absolute rounded cursor-pointer hover:brightness-125"
              style={{
                left: `${b.leftPct}%`,
                width: `${b.widthPct}%`,
                top: 3 + b.lane * rowHeight,
                height: barHeight,
                background: b.color
              }}
            />
          ))}
          <div className="absolute top-0 bottom-0 w-[2px] bg-[#f39c12]" style={{ left: `${todayPct}%` }} />
        </div>
      </div>
      <button onClick={() => setCreating(true)} className="text-xs bg-[var(--accent)] text-white rounded px-2 py-1 whitespace-nowrap">＋长期任务</button>

      {selected && (
        <div className="absolute z-50 top-12 right-40 bg-[var(--card)] border border-[var(--border)] rounded-lg p-3 shadow-xl w-56">
          <div className="font-bold text-sm mb-1">{selected.title}</div>
          <div className="text-xs text-[var(--muted)] mb-2">{formatCn(selected.start_date)} → {formatCn(selected.end_date)}（剩 {Math.max(0, diffDays(selected.end_date, today))} 天）</div>
          {selected.source_module && <div className="text-[10px] text-[var(--muted)] mb-2">来源：{selected.source_module}</div>}
          <div className="flex gap-2">
            <button onClick={() => { void remove(selected.id); setSelected(null) }} className="text-xs px-2 py-1 rounded bg-red-500/20 text-red-300">删除</button>
            <button onClick={() => setSelected(null)} className="text-xs px-2 py-1 rounded bg-[var(--border)]">关闭</button>
          </div>
        </div>
      )}

      {creating && (
        <div className="absolute z-50 top-12 right-40 bg-[var(--card)] border border-[var(--border)] rounded-lg p-3 shadow-xl w-64">
          <div className="font-bold text-sm mb-2">新建长期任务</div>
          <input autoFocus value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} placeholder="任务名称" className="w-full bg-[var(--bg)] border border-[var(--border)] rounded px-2 py-1 text-sm mb-2 outline-none" />
          <div className="flex gap-2 mb-3 text-xs text-[var(--muted)]">
            <input type="date" value={form.start} onChange={(e) => setForm({ ...form, start: e.target.value })} className="bg-[var(--bg)] border border-[var(--border)] rounded px-1 py-1" />
            <span>→</span>
            <input type="date" value={form.end} onChange={(e) => setForm({ ...form, end: e.target.value })} className="bg-[var(--bg)] border border-[var(--border)] rounded px-1 py-1" />
          </div>
          <div className="flex gap-2">
            <button onClick={() => void submitCreate()} className="text-xs px-2 py-1 rounded bg-[var(--accent)] text-white">创建</button>
            <button onClick={() => setCreating(false)} className="text-xs px-2 py-1 rounded bg-[var(--border)]">取消</button>
          </div>
        </div>
      )}
    </div>
  )
}
```

```tsx
import { useLongTasksStore } from '../../stores/longTasksStore'
import { todayStr, diffDays, formatCn } from '../../lib/dates'
import type { ReactNode } from 'react'

function ModuleCard({ icon, title, children }: { icon: string; title: string; children: ReactNode }): JSX.Element {
  return (
    <div className="bg-[var(--card)] border border-[var(--border)] rounded-lg p-3 mb-3">
      <div className="text-sm font-bold mb-2">{icon} {title}</div>
      {children}
    </div>
  )
}

export function TodayCards(): JSX.Element {
  const { tasks } = useLongTasksStore()
  const today = todayStr()
  const active = tasks.filter((t) => t.start_date <= today && t.end_date >= today)

  return (
    <div>
      <ModuleCard icon="🗓️" title="长期任务">
        {active.length === 0 && <div className="text-xs text-[var(--muted)]">暂无进行中的长期任务</div>}
        {active.map((t) => (
          <div key={t.id} className="flex items-center gap-2 text-xs py-1">
            <span className="w-2.5 h-2.5 rounded-sm" style={{ background: t.color }} />
            <span className="flex-1">{t.title}</span>
            <span className="text-[var(--muted)]">剩 {Math.max(0, diffDays(t.end_date, today))} 天</span>
          </div>
        ))}
      </ModuleCard>

      <ModuleCard icon="🏋️" title="健身">
        <div className="text-xs text-[var(--muted)]">健身模块将在 M4 接入</div>
      </ModuleCard>
      <ModuleCard icon="🍚" title="饮食">
        <div className="text-xs text-[var(--muted)]">饮食模块将在 M4 接入</div>
      </ModuleCard>
      <ModuleCard icon="📖" title="自我学习">
        <div className="text-xs text-[var(--muted)]">学习模块将在 M3 接入</div>
      </ModuleCard>
      <ModuleCard icon="💻" title="开发">
        <div className="text-xs text-[var(--muted)]">开发模块将在 M3 接入</div>
      </ModuleCard>
    </div>
  )
}
```
